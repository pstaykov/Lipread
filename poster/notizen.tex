\subsection{Vorverarbeitung}
\begin{itemize}
  \item MediaPipe FaceMesh (Google) liefert die Lippen-Landmarks. Quadrat mit Seitenlänge
        $\max(\text{Breite}, \text{Höhe}) \times 1{,}6$, zentriert auf die Mitte der Bounding-Box
        der Lippen.
  \item Glättung mit EMA, $\alpha = 0{,}4$ (der aktuelle Frame zählt 40\,\%, der Verlauf 60\,\%):
        \[
          \tilde{x}_t = \tfrac{1}{2}\bigl(\min_i x_{i,t} + \max_i x_{i,t}\bigr),\qquad
          x_t = \alpha\,\tilde{x}_t + (1-\alpha)\,x_{t-1},\quad \alpha = 0{,}4
        \]
        $\tilde{x}_t$ = rohe Mitte aus den Landmarks, $x_t$ = geglättete Mitte für den Zuschnitt.
        Nach ca.\ 5 Frames (0,2\,s) zählt ein Frame weniger als 8\,\%.
  \item Falls gefragt, was \enquote{Affin} bei der Augmentierung heißt: Rotation $\pm10^\circ$,
        Verschiebung $\pm6\,\%$, Skalierung $\pm10\,\%$. Einmal pro Clip gezogen und auf alle
        Frames gleich angewendet, nur im Training.
\end{itemize}

\section{Modell}

\subsection{Einordnung der Vorarbeiten}
\begin{itemize}
  \item \textbf{Martinez et al.\ (2020)} setzten mit dem MS-TCN den damaligen State of the Art auf
        LRW (BBC). Unser Frontend folgt ihrem Aufbau.
  \item \textbf{Schwiebert et al.\ (2022)} stellen GLips vor und zeigen Transfer zwischen LRW und
        GLips. X3D architektur. Verbesserungsbedarf
  \item \textbf{Ameer et al.}\ nutzen \textbf{NASNetMobile}: ein auf
        ImageNet vortrainiertes, kleines Bildnetz (4,7\,M Parameter), angewendet auf jeden Frame
        des ganzen Gesichts (128$\times$128, 16 Frames), dahinter ein GRU. Ergebnis: 48,4\,\%.
\end{itemize}

\phead{Warum wir NASNet nicht auf unsere Mund-Zuschnitte anwenden}
Wir vergleichen zeitliche Back-Ends (MS-TCN vs.Transformer)


\subsection{Architektur}
\begin{itemize}
  \item \textbf{Conv3D} ($5\times7\times7$) über Zeit, Höhe und Breite, RGB rein, 64 Kanäle raus.
        Danach MaxPool.
  \item \textbf{ResNet-18}, auf jedem Frame einzeln. Am Ende
        512 Kanäle $\times$ $3\times3$ Pixel pro Frame. Das kleine $3\times3$ ist gewollt: Das
        ResNet verdichtet die Information in die 512 Kanäle, darum geht beim räumlichen Pooling
        wenig verloren.
  \item \textbf{Räumliches Average-Pooling} $3\times3 \to 1$: Mittelwert über die 9 Positionen,
        keine lernbaren Gewichte.
  \item \textbf{Linear 512 $\to$ 256} + LayerNorm: legt die Arbeitsbreite fest (die
        Residualverbindungen brauchen gleiche Größe), hält den Transformer klein
        ($\approx$\,1,6\,M statt 6,3\,M Parameter) und normalisiert die Merkmale.
  \item \textbf{Multi-Scale-TCN}, Kernel 3, 5, 7 wie bei Martinez. Aber ohne Dilation und
        depthwise (siehe unten).
  \item \textbf{Transformer}: 8 Köpfe ist der Standardwert. Für eine Hyperparametersuche oder
        Kreuzvalidierung hatten wir keine Ressourcen. Die Köpfe können z.\,B.\ auf
        Mundöffnung, Vokal oder Nachbar-Frames achten.
  \item \textbf{FFN 1024}: Standard, $4\times$ die Breite 256. Kleines 2-Schicht-Netz pro Token
        innerhalb jeder Transformer-Schicht.

\end{itemize}

\subsubsection{Multi-Scale-TCN (in GNet)}
\begin{itemize}
  \item Drei Zweige parallel mit Kernel $k = 3, 5, 7$.
  \item \textbf{Depthwise:} Jeder der 256 Kanäle bekommt seinen eigenen Filter mit $k$ Gewichten
        über die Zeit. Padding $k/2$ (abgerundet), damit die 25 Frames erhalten bleiben.
  \item \textbf{1$\times$1-Convolution:} mischt danach die 256 Kanäle innerhalb eines Frames.
  \item Die drei Zweige werden \textbf{gemittelt} (nicht aneinandergehängt), dann Residual +
        LayerNorm. Ergebnis bleibt $25 \times 256$: ein Token pro Frame.
  \item Zwei Blöcke hintereinander: Sichtfeld 13 Frames ($\approx$\,0,5\,s). Den Rest des Clips
        verbindet der Transformer.
\end{itemize}

\subsubsection{Audio}
\begin{itemize}
  \item \textbf{Whisper-base-Encoder}, mehrsprachig, eingefroren. Nur der Encoder, also nur
        Audio-Merkmale, kein Sprachmodell.
  \item 1,16\,s Ton, mit Stille auf 2\,s aufgefüllt $\to$ 100 Tokens à 512.
  \item Whisper fügt vor seinen 6 Blöcken eigene sinusförmige Positions-Embeddings hinzu. Die
        Position steckt also schon im Inhalt der Tokens (als Information, keine magischen
        Zahlen). Für die Fusion: Linear auf 256 + Typ-Embedding (Video/Audio).
  \item \textbf{Nur Audio:} Mittelwert über die 100 Tokens $\to$ $1 \times 512$, LayerNorm,
        Linear 512 $\to$ 1024, GELU, Dropout 0,3, Linear 1024 $\to$ 500.
        \achtung{Whisper hat in seinen Embeddings schon globalen zeitlichen Kontext}
        (Self-Attention über alle Tokens), deshalb braucht es kein GNet-Back-End.
\end{itemize}

\subsubsection{Fusion}
\begin{itemize}
  \item Joint Transformer mit 3 Schichten: ein bewusst kleiner Fusionskopf auf bereits tiefen
        Merkmalen.
\end{itemize}

\section{Ergebnisse}
\begin{itemize}
  \item Gesunde Trainingskurven. Auf 500 Wörtern beginnt Overfitting (Training steigt weiter,
        Validierung flacht ab), deshalb Abbruch nach 15 Epochen.
  \item MS-TCN auf 500 Wörtern: ca 30\% leicht unter GNet. Lässt sich vermuten, dass auf mehr vokabular und mehr Daten GNet besser wird, da Transformer besser auf vielen Daten sind.

  \item \textbf{Transfer:} Alle Gewichte des 500-Wörter-Modells außer dem Klassifikator dienen als Initialisierung, dann wird alles weitertrainiert. Das 500er-Modell hat nur Trainingsclips gesehen.
  \item \textbf{Ausgegraut} im Vergleich mit Vorarbeiten:
        \begin{itemize}
          \item GNet + Transfer, weil es mit Vortraining auf allen 500 Wörtern nicht direkt mit den
                Vorarbeiten vergleichbar ist.
          \item Ameer (NASNet), weil es auf dem ganzen Gesicht ohne Mund-Zuschnitt arbeitet. Wir

        \end{itemize}
\end{itemize}

\subsection{Ablationen}
\begin{itemize}
  \item \textbf{Transformer vs.\ MS-TCN:} kein messbarer Vorteil. Beide sehen den ganzen Clip, und der ganze zeitliche Kontext reicht. Deshalb vermutlich auch nur ein kleiner Vorteil für DC-TCN (nur mit Ensemble). Auf mehr Vokabular und mehr Daten lässt sich ein Vorteil von GNet erahnen.
  \item \textbf{Mean-Pooling reicht}
  \item \textbf{Ohne Multi-Scale-Stem:} etwas schlechter (56,9 statt 59,3), selbsterklärend.
  \item \textbf{Ohne Mund-ROI:} $-29$ Punkte, die Lippen sind das Wichtigste.
\end{itemize}

\section{Diskussion}
\begin{itemize}
  \item Alle Aufnahmen stammen aus einem Landtag: frontal, gut ausgeleuchtet, erwachsene Sprecher.
  \item Ähnliche Wörter wie „herrn“, „einmal“ und „linke“ untrennbar.
  \item Sprecher überschneiden sich zwischen Train und Test (keine Sprecher-Labels in GLips).
  \item Auf 15 Wörter merk sich das Modell die Länge nicht das Wort
  \item Das Vokabular ist auf 500 Einzelwörter begrenzt, ganze Sätze werden nicht erkannt.

\end{itemize}
